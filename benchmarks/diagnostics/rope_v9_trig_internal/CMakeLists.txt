cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(rope_v9_trig_internal LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 17)
set(ROPE_DIAG_MATH_DIR "" CACHE PATH "Generated source-bound V9 math snapshot")
add_executable(trig_internal_probe trig_internal_probe.asc)
target_include_directories(trig_internal_probe PRIVATE ${ROPE_DIAG_MATH_DIR})
target_link_libraries(trig_internal_probe PRIVATE ascendcl m)
target_compile_options(trig_internal_probe PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=dav-2201>)
target_compile_definitions(trig_internal_probe PRIVATE ROPE_LAUNCH_AUDIT=1)
target_link_options(trig_internal_probe PRIVATE -Wl,--wrap=aclrtLaunchKernelWithHostArgs)
