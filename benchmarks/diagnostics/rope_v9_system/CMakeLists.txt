cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(rope_v9_stage_diagnostic LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 17)
set(ROPE_DIAG_MATH_DIR "" CACHE PATH "Generated unmodified V9 math helpers")
add_executable(stage_probe stage_probe.asc)
target_include_directories(stage_probe PRIVATE ${ROPE_DIAG_MATH_DIR})
target_link_libraries(stage_probe PRIVATE ascendcl m)
target_compile_options(stage_probe PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=dav-2201>)
target_compile_definitions(stage_probe PRIVATE ROPE_LAUNCH_AUDIT=1)
target_link_options(stage_probe PRIVATE -Wl,--wrap=aclrtLaunchKernelWithHostArgs)
